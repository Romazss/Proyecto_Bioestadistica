\begin{frame}{El artículo analizado}

\section{P1 — IA vs.\ Estadística}
% ============================================================

\begin{frame}{Pregunta 1}
  \framesubtitle{¿Qué distingue conceptualmente a la IA de la Estadística en este artículo?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Estadística}
        \begin{itemize}
          \item Inferir asociaciones e \textbf{interpretar} relaciones entre variables.
          \item Correlación de Spearman, mapas de calor.
          \item Interpretabilidad \textbf{directa}: cada coeficiente revela dirección y fuerza.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{IA / Machine Learning}
        \begin{itemize}
          \item Priorizar la \textbf{predicción} a partir de datos históricos.
          \item Random Forest, XGBoost: capturan relaciones no lineales.
          \item Operan como \textbf{caja negra}: baja interpretabilidad.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{Marco teórico de referencia}
    \begin{itemize}
      \item \textbf{Breiman (2001):} \textit{dos culturas} — modelo de datos vs.\ modelo algorítmico.
      \item \textbf{Shmueli (2010):} \textit{explanatory modeling} vs.\ \textit{predictive modeling}.
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{XAI como puente entre las dos culturas}
  \framesubtitle{Pregunta 1 — Conclusión}
  \begin{center}
  \begin{tikzpicture}[
      pill/.style={rectangle, rounded corners=8pt, fill=#1, text width=3.2cm,
                   align=center, minimum height=1.1cm, font=\small\bfseries},
      arr/.style={<->, very thick, color=PUCgold}
    ]
    \node[pill=PUCblue, text=white] (est) at (0,0) {Estadística\\{\normalfont Interpretar}};
    \node[pill=PUCblue, text=white] (ia)  at (8,0) {IA / ML\\{\normalfont Predecir}};
    \node[pill=PUCgold, text=PUCblue] (xai) at (4,0) {XAI\\(SHAP)\\{\normalfont Puente}};
    \draw[arr] (est) -- (xai);
    \draw[arr] (xai) -- (ia);
  \end{tikzpicture}
  \end{center}
  \vspace{0.5em}
  \begin{itemize}
    \item La distinción es \textbf{funcional}, no esencial: la Regresión Logística es a la vez
          ML y estadística clásica.
    \item \textbf{SHAP} (basado en la teoría de juegos cooperativos de Shapley) reintroduce
          interpretabilidad en el modelo predictivo.
    \item Tang et al.\ (2024) trata la IA como una \textbf{extensión predictiva} de la
          Estadística, no como su antítesis.
  \end{itemize}
\end{frame}

\section{P2 — Predicción vs.\ Explicación}
% ============================================================

\begin{frame}{Pregunta 2 — Punto de partida}
  \framesubtitle{¿El artículo distingue entre predicción y explicación?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Predecir (RAE, def.\ 1)}
        «Anunciar por \textbf{conocimiento fundado} algo que ha de suceder.»
      \end{exampleblock}
      \vspace{0.5em}
      \begin{exampleblock}{Explicar — def.\ 2 (expositiva)}
        «Declarar una materia con palabras claras para hacerla \textbf{más perceptible}.»
        \vspace{0.3em}

        \textit{Bajo esta definición: predicción $\approx$ explicación}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Explicar — def.\ 5 (causal)}
        «Dar a conocer la \textbf{causa o motivo} de algo.»
        \vspace{0.3em}

        \textit{Bajo esta definición: predicción $\neq$ explicación}
      \end{alertblock}
      \vspace{0.5em}
      \begin{block}{SHAP explica el \textit{modelo}, no el fenómeno}
        Atribuye contribuciones a las variables \textbf{dentro del modelo}.
        No establece mecanismos causales del suicidio.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Predicción $\neq$ Explicación causal}
  \framesubtitle{Pregunta 2 — Marco de Shmueli (2010) y Pearl (2009)}
  \begin{itemize}
    \item \textbf{Shmueli (2010):} el artículo se inscribe en la cultura \textit{predictiva}:
          su criterio de éxito es el desempeño ($\text{AUC} = 97{,}05\,\%$), no la verificación
          de una hipótesis causal.
    \medskip
    \item \textbf{Pearl (2009):} la correlación no implica causalidad.
          Sin un diseño que controle confusores, no es posible elevar una
          asociación predictiva a relación causal.
  \end{itemize}
  \vspace{0.5em}
  \begin{alertblock}{Consecuencia práctica}
    Que \textbf{ira}, \textbf{depresión} y \textbf{aislamiento social} sean los predictores más
    importantes (Tang et al., 2024, Fig.\ 6b) \textbf{no equivale a afirmar que los causen}.
    Podrían ser síntomas, consecuencias o efectos de un confusor común.
  \end{alertblock}
  \vspace{0.4em}
  \begin{block}{Conclusión de la P2}
    \small
    Bajo definición expositiva: predicción $\approx$ explicación.
    Bajo definición causal: \textbf{difieren}. El aporte del artículo es predictivo;
    la XAI cumple una función interpretativa \textit{sobre el modelo}, no causal \textit{del fenómeno}.
  \end{block}
\end{frame}

\section{P3 — Validez del patrón}
% ============================================================

\begin{frame}{Pregunta 3 — Evidencia presentada}
  \framesubtitle{¿Qué justifica que el patrón corresponde al fenómeno y no a un artefacto?}
  \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Evidencia descriptiva}
        \begin{itemize}
          \item \textbf{Conclusión 1:} nube de palabras de los registros\\
                $\Rightarrow$ asociación con trastornos.
          \item \textbf{Conclusión 2:} gráfico de barras\\
                $\Rightarrow$ desempleo = mayor riesgo;\\
                personal de seguridad = menor riesgo.
          \item Coincidencia con resultados de otros estudios.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Evidencia del modelo}
        \begin{itemize}
          \item Correlación de Spearman + mapa de calor.
          \item Random Forest + SHAP\\
                $\Rightarrow$ ira, depresión, aislamiento social.
          \item Aumento de datos (CTGAN, SMOTENC)\\
                $\Rightarrow$ mayor robustez interna.
          \item Factores protectores: alto nivel educativo.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{alertblock}{Limitación central}
    Todo el análisis usa la \textbf{misma base de datos}.
    El aumento de datos (CTGAN/SMOTENC) sintetiza registros a partir de
    la distribución original: las atribuciones SHAP describen \textbf{el modelo},
    no el mecanismo del fenómeno (Breiman, 2001; Shmueli, 2010; Pearl, 2009).
  \end{alertblock}
\end{frame}

\begin{frame}{Pregunta 3 — Conclusión}
  \framesubtitle{Alcance de la evidencia}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\faCheckCircle\  Lo que sí respalda la evidencia}
        \begin{itemize}
          \item El patrón es \textbf{estable} dentro del conjunto de datos.
          \item Es \textbf{útil para predecir} en el mismo dominio.
          \item Coincide con hallazgos previos de la literatura.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\faExclamationTriangle\ Lo que queda abierto}
        \begin{itemize}
          \item Justificación \textbf{teórico-causal} más allá de los datos.
          \item Perspectiva \textbf{psicológica/psiquiátrica} ausente.
          \item Generalización a otros contextos o poblaciones.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{El propio artículo reconoce la limitación}
    \centering
    \small
    «Los modelos de predicción deben utilizarse \textbf{en conjunto con el
    asesoramiento médico}.»\\
    \textit{(Tang et al., 2024, Clinical Implications and Future Directions)}
  \end{block}
\end{frame}

\section{Conclusiones}
% ============================================================

\begin{frame}{Conclusiones generales}
  \begin{enumerate}
    \item[\textbf{P1}] IA y Estadística \textbf{no se oponen} en Tang et al.\ (2024):
          se distinguen por su énfasis (predecir vs.\ interpretar) y la XAI actúa como
          mecanismo de traducción entre ambas.
    \vspace{0.5em}
    \item[\textbf{P2}] La respuesta depende de la definición de ``explicar'':
          bajo la definición \textit{causal}, predicción y explicación \textbf{difieren}.
          SHAP explica el modelo, no el fenómeno.
    \vspace{0.5em}
    \item[\textbf{P3}] La evidencia justifica que el patrón es \textbf{estable y predictivo}
          dentro del marco de datos, pero la justificación teórico-causal del fenómeno
          queda abierta.
  \end{enumerate}
  \vspace{0.6em}
  \begin{block}{Síntesis}
    \centering
    \small
    Los modelos de ML proveen una explicación de los registros obtenidos y, por ello,
    deben usarse como \textbf{complemento y validación} del juicio médico, no como sustituto.
  \end{block}
\end{frame}
